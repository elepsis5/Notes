case 'search':
    // dd($this->filters);
    $value = array_key_exists($code, $this->filters) ? $this->filters[$code] : null;
    if ($value === null || ! is_string($value)) break;

    // раскодирывание html сущностей в цикле (обратное htmlspecialchars)
    do {
        $prev = $value;
        $value = html_entity_decode($value, ENT_QUOTES | ENT_HTML5, 'UTF-8');
    } while ($value !== $prev && str_contains($value, '&'));

    // убираем html теги
    $value = strip_tags($value);
    // заменяем ряд символов на пробелы
    $value = preg_replace('/(\(|\)|\.|\/)/', ' ', $value);
    // убираем всё кроме букв, цифр, пробелов
    $value = trim(preg_replace('/[^\p{L}\p{N}\s]/u', '', $value));
    // заменяем пробелы на regex оператор "или"
    $value = preg_replace('/\s+/', '|', $value);
